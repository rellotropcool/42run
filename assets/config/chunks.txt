# =============================================================================
#  CHUNK DEFINITIONS
# =============================================================================
#
#  One chunk per line.
#
#    chunk    : <lane>,<type>,<zSlot>  [<lane>,<type>,<zSlot> ...]
#    breather : <length>
#
#  lane  = 0 (left), 1 (middle), 2 (right)
#  type  = f  full block  — must dodge, too tall to jump
#          s  semi block  — can dodge OR jump over
#          t  top block   — must slide under
#  zSlot = forward offset within this chunk (0, 1, 2, ...)
#
#  Chunk length = highest zSlot + 1.
#  Duplicate a line to weight it more heavily.
#
#  Coins are NOT placed here — CoinSpawner streams them independently.
#  Powerups are NOT placed here — PowerupSpawner spawns them periodically.
#  This file is obstacles only.
#
# =============================================================================
